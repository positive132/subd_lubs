SELECT LAST_NAME  FROM STUDENT 
WHERE LAST_NAME LIKE '%б%' OR LAST_NAME LIKE '%о%';

SELECT * FROM STUDENT 
WHERE LAST_NAME LIKE 'К%' AND S_NAME IS NULL;

SELECT * FROM STUDENT 
WHERE LEN(LAST_NAME)>=8;

SELECT * FROM STUDENT 
WHERE LEN(LAST_NAME)<>7 AND LAST_NAME LIKE '%а%';

SELECT * FROM STUDENT 
WHERE FACULTY LIKE 'ФПМ' AND FORM LIKE 'очная' AND YEARR IN (1, 2)
ORDER BY S_NAME;

SELECT * FROM STUDENT 
WHERE FACULTY LIKE 'ФПК' AND FORM LIKE 'заочная' AND EXM >6
ORDER BY EXM DESC;

SELECT * FROM TEACHER 
WHERE FACULTY LIKE 'ФПК'
ORDER BY FORM,LAST_NAME;

SELECT * FROM TEACHER 
WHERE FACULTY LIKE 'ФПМ' AND YEARR=1 AND HOURSS>100;

SELECT * FROM TEACHER 
WHERE S_NAME IS NULL AND DATEDIFF(YEAR, START_WORK_DATE, GETDATE())>3;

SELECT DISTINCT SUBJ FROM TEACHER 
WHERE FACULTY LIKE 'ФПМ' AND YEARR=3;

SELECT YEARR,FORM,LAST_NAME,F_NAME,S_NAME,SUBJ,HOURSS FROM TEACHER 
WHERE FACULTY LIKE 'ФПК' AND HOURSS >100;

SELECT FACULTY, YEARR, FORM, LAST_NAME, F_NAME, S_NAME, SUBJ FROM TEACHER 
WHERE S_NAME IS NULL;

SELECT LAST_NAME,F_NAME,S_NAME FROM TEACHER
WHERE DATEDIFF(YEAR,BR_DATE,CONCAT(YEAR(GETDATE()), '0101'))>30;

SELECT LAST_NAME,F_NAME,S_NAME  FROM TEACHER 
WHERE DATEDIFF(YEAR, BR_DATE, GETDATE()) BETWEEN 35 AND 40
ORDER BY LAST_NAME;

SELECT LAST_NAME,F_NAME,S_NAME FROM TEACHER 
WHERE MONTH(BR_DATE) = 10
ORDER BY BR_DATE ASC;

SELECT FACULTY, AVG(EXM) AS AVERAGE_EXM FROM STUDENT
WHERE FORM LIKE 'заочная' GROUP BY FACULTY;

SELECT FACULTY,YEARR, MAX(EXM) AS MAX_EXM FROM STUDENT
GROUP BY FACULTY,YEARR;

SELECT FACULTY, AVG(EXM) AS AVERAGE_EXM FROM STUDENT
GROUP BY FACULTY HAVING AVG(EXM) >7

SELECT YEARR,FACULTY,FORM, AVG(EXM) AS AVERAGE_EXM FROM STUDENT
GROUP BY YEARR,FACULTY,FORM HAVING AVG(EXM) >7.5;

SELECT FACULTY, YEARR, MIN(EXM) AS MIN_EXM FROM STUDENT
GROUP BY FACULTY, YEARR;

SELECT FACULTY, FORM, MIN(EXM) AS MIN_EXM FROM STUDENT
GROUP BY FACULTY, FORM HAVING MIN(EXM) >6;

SELECT AVG(ALL_H-INCLASS_H) AS SAMPO FROM STUDENT
WHERE FACULTY LIKE 'ФПК' AND YEARR LIKE 3 AND FORM LIKE 'заочная';

SELECT FACULTY,YEARR,FORM, AVG(ALL_H - INCLASS_H) AS SAMPO FROM STUDENT
GROUP BY FACULTY,YEARR,FORM HAVING AVG(ALL_H - INCLASS_H) >150;

SELECT LAST_NAME,F_NAME,S_NAME, COUNT(DISTINCT SUBJ) AS SUBCOUNT FROM TEACHER
GROUP BY LAST_NAME,F_NAME,S_NAME;

SELECT FACULTY, COUNT(DISTINCT CONCAT(LAST_NAME,F_NAME)) AS TEACHCOUNT FROM TEACHER
GROUP BY FACULTY;

SELECT SUBJ, MAX(HOURSS) AS MAX_HOUR FROM TEACHER
GROUP BY SUBJ;

SELECT LAST_NAME,F_NAME,S_NAME FROM TEACHER
GROUP BY LAST_NAME,F_NAME,S_NAME
HAVING COUNT(DISTINCT SUBJ) >1;

SELECT FACULTY,YEARR, SUM(HOURSS) SUM_HOUR FROM TEACHER
GROUP BY FACULTY,YEARR;

SELECT FACULTY,YEARR, COUNT(DISTINCT SUBJ) AS SUBCOUNT FROM TEACHER
WHERE YEARR=2
GROUP BY FACULTY, YEARR ORDER BY FACULTY DESC, YEARR ASC;

SELECT FACULTY, COUNT(DISTINCT SUBJ) AS SUBCOUNT FROM TEACHER
WHERE S_NAME IS NULL
GROUP BY FACULTY;
