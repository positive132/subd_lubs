CREATE TABLE Departments (
    DeptID INT PRIMARY KEY IDENTITY(1,1),
    DeptName NVARCHAR(50) NOT NULL
);

CREATE TABLE EduForms (
    FormID INT PRIMARY KEY IDENTITY(1,1),
    FormName NVARCHAR(50) NOT NULL
);

CREATE TABLE AcademicPlans (
    PlanID INT PRIMARY KEY IDENTITY(1,1),
    StudyYear INT NOT NULL,
    DeptID INT NOT NULL,
    FormID INT NOT NULL,
    TotalHours INT,
    ClassroomHours INT,
    CONSTRAINT FK_Plans_Departments FOREIGN KEY (DeptID) REFERENCES Departments(DeptID),
    CONSTRAINT FK_Plans_EduForms FOREIGN KEY (FormID) REFERENCES EduForms(FormID)
);

CREATE TABLE Enrollees (
    EnrolleeID INT PRIMARY KEY IDENTITY(1,1),
    Surname NVARCHAR(25) NOT NULL,
    FirstName NVARCHAR(25) NOT NULL,
    Patronymic NVARCHAR(25),
    BirthDate DATE,
    AdmissionDate DATE,
    ScoreValue FLOAT
);

CREATE TABLE Enrollments (
    EnrolleeID INT NOT NULL,
    PlanID INT NOT NULL,
    CONSTRAINT PK_Enrollments PRIMARY KEY (EnrolleeID, PlanID),
    CONSTRAINT FK_Enrollments_Enrollees FOREIGN KEY (EnrolleeID) REFERENCES Enrollees(EnrolleeID),
    CONSTRAINT FK_Enrollments_Plans FOREIGN KEY (PlanID) REFERENCES AcademicPlans(PlanID)
);

SELECT e.Surname, e.FirstName, e.Patronymic, d.DeptName, p.StudyYear FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE f.FormName = 'Заочная' AND FLOOR(DATEDIFF(DAY, e.BirthDate, GETDATE()) / 365.25) < 37;

SELECT d.DeptName, COUNT(en.EnrolleeID) AS ResultCount FROM Departments d
JOIN AcademicPlans p ON d.DeptID = p.DeptID
JOIN Enrollments en ON p.PlanID = en.PlanID
GROUP BY d.DeptName;

SELECT f.FormName, COUNT(en.EnrolleeID) AS ResultCount FROM EduForms f
JOIN AcademicPlans p ON f.FormID = p.FormID
JOIN Enrollments en ON p.PlanID = en.PlanID
GROUP BY f.FormName;

SELECT d.DeptName, AVG(FLOOR(DATEDIFF(DAY, e.BirthDate, GETDATE()) / 365.25)) AS MeanAge FROM Departments d
JOIN AcademicPlans p ON d.DeptID = p.DeptID
JOIN Enrollments en ON p.PlanID = en.PlanID
JOIN Enrollees e ON en.EnrolleeID = e.EnrolleeID
GROUP BY d.DeptName;

SELECT e.Surname, e.FirstName, e.AdmissionDate, d.DeptName, p.StudyYear, f.FormName FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE e.Patronymic IS NULL;

SELECT d.DeptName, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
WHERE YEAR(e.AdmissionDate) = 2015
GROUP BY d.DeptName
ORDER BY ResultCount DESC;

SELECT d.DeptName, f.FormName, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE YEAR(e.AdmissionDate) = 2014
GROUP BY d.DeptName, f.FormName;

SELECT DISTINCT d.DeptName FROM Departments d
JOIN AcademicPlans p ON d.DeptID = p.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE f.FormName = 'Заочная';

SELECT d.DeptName, f.FormName, p.StudyYear FROM Departments d
JOIN AcademicPlans p ON d.DeptID = p.DeptID
JOIN EduForms f ON p.FormID = f.FormID
ORDER BY d.DeptName, f.FormName, p.StudyYear;

SELECT d.DeptName, f.FormName, COUNT(en.EnrolleeID) AS ResultCount FROM Departments d
JOIN AcademicPlans p ON d.DeptID = p.DeptID
JOIN EduForms f ON p.FormID = f.FormID
JOIN Enrollments en ON p.PlanID = en.PlanID
GROUP BY d.DeptName, f.FormName
ORDER BY d.DeptName;

SELECT d.DeptName, f.FormName, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE p.StudyYear IN (1, 3)
GROUP BY d.DeptName, f.FormName;

SELECT d.DeptName, p.StudyYear, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
WHERE e.Patronymic IS NULL
GROUP BY d.DeptName, p.StudyYear
ORDER BY d.DeptName, p.StudyYear;

SELECT d.DeptName, p.StudyYear, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
WHERE e.ScoreValue > 7.5
GROUP BY d.DeptName, p.StudyYear
ORDER BY d.DeptName, p.StudyYear;

SELECT d.DeptName, f.FormName, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE FLOOR(DATEDIFF(DAY, e.BirthDate, GETDATE()) / 365.25) > 45
GROUP BY d.DeptName, f.FormName
ORDER BY d.DeptName;

SELECT d.DeptName, f.FormName, p.StudyYear, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
JOIN EduForms f ON p.FormID = f.FormID
WHERE FLOOR(DATEDIFF(DAY, e.BirthDate, GETDATE()) / 365.25) < 27
GROUP BY d.DeptName, f.FormName, p.StudyYear
ORDER BY d.DeptName, f.FormName, p.StudyYear;

SELECT d.DeptName, COUNT(e.EnrolleeID) AS ResultCount FROM Enrollees e
JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
JOIN AcademicPlans p ON en.PlanID = p.PlanID
JOIN Departments d ON p.DeptID = d.DeptID
WHERE e.Surname LIKE 'С%'
GROUP BY d.DeptName
ORDER BY d.DeptName;
