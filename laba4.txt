CREATE TABLE Departments (
    DeptID INT PRIMARY KEY IDENTITY(1,1),
    DeptName NVARCHAR(50) NOT NULL
);

CREATE TABLE EduForms (
    FormID INT PRIMARY KEY IDENTITY(1,1),
    FormName NVARCHAR(50) NOT NULL
);

CREATE TABLE AcademicPlans (
    PlanID INT PRIMARY KEY IDENTITY(1,1),
    StudyYear INT NOT NULL,
    DeptID INT NOT NULL,
    FormID INT NOT NULL,
    TotalHours INT,
    ClassroomHours INT,
    CONSTRAINT FK_Plans_Departments FOREIGN KEY (DeptID) REFERENCES Departments(DeptID),
    CONSTRAINT FK_Plans_EduForms FOREIGN KEY (FormID) REFERENCES EduForms(FormID)
);

CREATE TABLE Enrollees (
    EnrolleeID INT PRIMARY KEY IDENTITY(1,1),
    Surname NVARCHAR(25) NOT NULL,
    FirstName NVARCHAR(25) NOT NULL,
    Patronymic NVARCHAR(25),
    BirthDate DATE,
    AdmissionDate DATE,
    ScoreValue FLOAT
);

CREATE TABLE Enrollments (
    EnrolleeID INT NOT NULL,
    PlanID INT NOT NULL,
    CONSTRAINT PK_Enrollments PRIMARY KEY (EnrolleeID, PlanID),
    CONSTRAINT FK_Enrollments_Enrollees FOREIGN KEY (EnrolleeID) REFERENCES Enrollees(EnrolleeID),
    CONSTRAINT FK_Enrollments_Plans FOREIGN KEY (PlanID) REFERENCES AcademicPlans(PlanID)
);

CREATE PROCEDURE GetStudentsCountByFilters
    @TargetDept NVARCHAR(50),
    @TargetForm NVARCHAR(50)
AS
BEGIN
    SELECT COUNT(e.EnrolleeID) AS ResultCount 
    FROM Enrollees e
    JOIN Enrollments en ON e.EnrolleeID = en.EnrolleeID
    JOIN AcademicPlans p ON en.PlanID = p.PlanID
    JOIN Departments d ON p.DeptID = d.DeptID
    JOIN EduForms f ON p.FormID = f.FormID 
    WHERE d.DeptName = @TargetDept 
      AND f.FormName = @TargetForm;
END;
GO

CREATE PROCEDURE CalculatePlanMetrics
    @OutFpkCount INT OUTPUT,
    @OutFpmCount INT OUTPUT,
    @OutTotalCount INT OUTPUT,
    @OutDuplicateCount INT OUTPUT
AS
BEGIN
    SELECT @OutFpkCount = COUNT(*) 
    FROM AcademicPlans p
    JOIN Departments d ON p.DeptID = d.DeptID 
    WHERE d.DeptName = 'ФПК';

    SELECT @OutFpmCount = COUNT(*) 
    FROM AcademicPlans p
    JOIN Departments d ON p.DeptID = d.DeptID 
    WHERE d.DeptName = 'ФПМ';

    SELECT @OutTotalCount = COUNT(*) 
    FROM AcademicPlans;

    SELECT @OutDuplicateCount = COUNT(*) 
    FROM (
        SELECT StudyYear 
        FROM AcademicPlans 
        GROUP BY StudyYear 
        HAVING COUNT(*) > 1
    ) AS DuplicateGroups;

    PRINT 'ФПК: ' + CAST(@OutFpkCount AS NVARCHAR(10));
    PRINT 'ФПМ: ' + CAST(@OutFpmCount AS NVARCHAR(10));
    PRINT 'Всего: ' + CAST(@OutTotalCount AS NVARCHAR(10));
    PRINT 'Одинаковых: ' + CAST(@OutDuplicateCount AS NVARCHAR(10));
END;
GO

CREATE PROCEDURE InsertNewEnrollee
    @TxtLastName NVARCHAR(25),
    @TxtFirstName NVARCHAR(25),
    @TxtPatronymic NVARCHAR(25) = NULL,
    @DtBirth DATE,
    @DtAdmission DATE,
    @ParamDept NVARCHAR(50),
    @ParamForm NVARCHAR(50)
AS
BEGIN
    DECLARE @FoundPlanID INT;

    SELECT @FoundPlanID = p.PlanID 
    FROM AcademicPlans p
    JOIN Departments d ON p.DeptID = d.DeptID
    JOIN EduForms f ON p.FormID = f.FormID 
    WHERE d.DeptName = @ParamDept 
      AND f.FormName = @ParamForm 
      AND p.StudyYear = 1;

    INSERT INTO Enrollees (Surname, FirstName, Patronymic, BirthDate, AdmissionDate, ScoreValue) 
    VALUES (@TxtLastName, @TxtFirstName, @TxtPatronymic, @DtBirth, @DtAdmission, 0);
    DECLARE @CreatedStudentID INT = SCOPE_IDENTITY();
    INSERT INTO Enrollments (EnrolleeID, PlanID) 
    VALUES (@CreatedStudentID, @FoundPlanID);
    PRINT N'Студент добавлен';
END;
GO
